create table public.agents (
  id bigint not null,
  sp_id integer not null,
  mini_fen text not null,
  color character(1) not null,
  current_elo integer not null default 2000,
  constraint agents_pkey primary key (id),
  constraint unique_sp_color unique (sp_id, color),
  constraint agents_sp_id_check check (
    (
      (sp_id >= 0)
      and (sp_id < 960)
    )
  ),
  constraint agents_color_check check ((color = any (array['w'::bpchar, 'b'::bpchar]))),
  constraint agents_mini_fen_len_chk check ((char_length(mini_fen) >= 8))
) TABLESPACE pg_default;

create table public.championships (
  id bigint generated always as identity not null,
  name text not null,
  description text null,
  start_date timestamp with time zone not null default now(),
  end_date timestamp with time zone null,
  status character varying(20) not null default 'active'::character varying,
  created_at timestamp with time zone not null default now(),
  updated_at timestamp with time zone not null default now(),
  total_games_expected bigint not null default 921600,
  notes jsonb null,
  constraint championships_pkey primary key (id),
  constraint championships_status_check check (
    (
      (status)::text = any (
        array[
          ('pending'::character varying)::text,
          ('active'::character varying)::text,
          ('completed'::character varying)::text,
          ('archived'::character varying)::text
        ]
      )
    )
  )
) TABLESPACE pg_default;

create index IF not exists idx_championships_status on public.championships using btree (status) TABLESPACE pg_default;

create index IF not exists idx_championships_dates on public.championships using btree (start_date, end_date) TABLESPACE pg_default;

create table public.elo_history (
  id bigserial not null,
  player_id bigint not null,
  championship_id bigint not null,
  match_id bigint not null,
  elo_before integer not null,
  elo_after integer not null,
  elo_change integer not null,
  created_at timestamp with time zone not null default now(),
  constraint elo_history_pkey primary key (id),
  constraint unique_player_match unique (player_id, match_id),
  constraint elo_history_player_id_fkey foreign KEY (player_id) references agents (id) on delete CASCADE,
  constraint elo_history_championship_id_fkey foreign KEY (championship_id) references championships (id) on delete CASCADE,
  constraint elo_history_match_id_fkey foreign KEY (match_id) references matches (id) on delete CASCADE
) TABLESPACE pg_default;

create index IF not exists idx_elo_history_player on public.elo_history using btree (player_id) TABLESPACE pg_default;

create index IF not exists idx_elo_history_championship on public.elo_history using btree (championship_id) TABLESPACE pg_default;

create index IF not exists idx_elo_history_match on public.elo_history using btree (match_id) TABLESPACE pg_default;

create index IF not exists idx_elo_history_created_at on public.elo_history using btree (created_at) TABLESPACE pg_default;

create index IF not exists idx_elo_history_player_champ on public.elo_history using btree (player_id, championship_id) TABLESPACE pg_default;

create index IF not exists elo_history_player_created_idx on public.elo_history using btree (player_id, created_at desc) TABLESPACE pg_default;

create table public.matches (
  id bigint generated by default as identity not null,
  created_at timestamp with time zone not null default now(),
  player_white bigint not null,
  player_black bigint not null,
  pgn text null,
  status boolean not null default false,
  round bigint null,
  championship_id bigint not null,
  result text null,
  claimed_at timestamp with time zone null,
  constraint matches_pkey primary key (id),
  constraint matches_player_white_fkey foreign KEY (player_white) references agents (id) on delete CASCADE,
  constraint matches_player_black_fkey foreign KEY (player_black) references agents (id) on delete CASCADE,
  constraint matches_championship_id_fkey foreign KEY (championship_id) references championships (id) on delete CASCADE
) TABLESPACE pg_default;

create index IF not exists idx_matches_player_white on public.matches using btree (player_white) TABLESPACE pg_default;

create index IF not exists idx_matches_player_black on public.matches using btree (player_black) TABLESPACE pg_default;

create index IF not exists idx_matches_players_pair on public.matches using btree (player_white, player_black) TABLESPACE pg_default;

create index IF not exists idx_matches_championship on public.matches using btree (championship_id) TABLESPACE pg_default;

create index IF not exists idx_matches_championship_round on public.matches using btree (championship_id, round) TABLESPACE pg_default;

create index IF not exists idx_matches_claimable on public.matches using btree (id) TABLESPACE pg_default
where
  (
    (status = false)
    and (claimed_at is null)
  );

create index IF not exists matches_finished_desc_idx on public.matches using btree (id desc) TABLESPACE pg_default
where
  (
    (status = true)
    and (result is not null)
  );

create table public.standings (
  id bigint generated by default as identity not null,
  player_id bigint not null,
  games_played integer not null default 0,
  wins integer not null default 0,
  draws integer not null default 0,
  losses integer not null default 0,
  points numeric(5, 2) not null default 0.0,
  bergvizer_score numeric(7, 3) null default null::numeric,
  championship_id bigint not null,
  constraint standings_current_pkey primary key (id),
  constraint standings_current_player_id_fkey foreign KEY (player_id) references agents (id) on delete CASCADE,
  constraint standings_championship_id_fkey foreign KEY (championship_id) references championships (id) on delete CASCADE
) TABLESPACE pg_default;

create index IF not exists idx_standings_championship on public.standings using btree (championship_id) TABLESPACE pg_default;
