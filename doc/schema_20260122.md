create table public.sf_agent_favorites (
  id uuid not null default gen_random_uuid (),
  user_id uuid not null,
  agent_id integer not null,
  created_at timestamp with time zone null default now(),
  constraint sf_agent_favorites_pkey primary key (id),
  constraint sf_agent_favorites_user_id_agent_id_key unique (user_id, agent_id),
  constraint sf_agent_favorites_user_id_fkey foreign KEY (user_id) references auth.users (id) on delete CASCADE,
  constraint sf_agent_favorites_agent_id_fkey foreign KEY (agent_id) references sf_agents (id) on delete CASCADE
) TABLESPACE pg_default;

-- public.sf_agents

create table public.sf_agents (
  id bigint not null,
  sp_id integer not null,
  mini_fen text not null,
  color character(1) not null,
  current_elo integer not null default 2000,
  constraint sf_agents_pkey primary key (id),
  constraint sf_agents_sp_id_color_key unique (sp_id, color),
  constraint agents_color_check check ((color = any (array['w'::bpchar, 'b'::bpchar]))),
  constraint agents_mini_fen_len_chk check ((char_length(mini_fen) >= 8)),
  constraint agents_sp_id_check check (
    (
      (sp_id > 0)
      and (sp_id <= 960)
    )
  )
) TABLESPACE pg_default;

-- public.sf_championships

create table public.sf_championships (
  id bigint generated always as identity not null,
  name text not null,
  description text null,
  start_date timestamp with time zone not null default now(),
  end_date timestamp with time zone null,
  status character varying(20) not null default 'active'::character varying,
  created_at timestamp with time zone not null default now(),
  updated_at timestamp with time zone not null default now(),
  total_games_expected bigint not null default 921600,
  notes jsonb null,
  constraint sf_championships_pkey primary key (id),
  constraint championships_status_check check (
    (
      (status)::text = any (
        array[
          ('pending'::character varying)::text,
          ('active'::character varying)::text,
          ('completed'::character varying)::text,
          ('archived'::character varying)::text
        ]
      )
    )
  )
) TABLESPACE pg_default;

create index IF not exists sf_championships_status_idx on public.sf_championships using btree (status) TABLESPACE pg_default;

create index IF not exists sf_championships_start_date_end_date_idx on public.sf_championships using btree (start_date, end_date) TABLESPACE pg_default;

-- 2. Создаём функцию для триггера
CREATE OR REPLACE FUNCTION public.trigger_set_timestamp()
RETURNS TRIGGER AS $$
BEGIN
  NEW.updated_at = NOW();
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- 3. Создаём триггер
create trigger set_timestamp
before update on public.profiles
for each row
execute function trigger_set_timestamp();

-- 4. Уникальный индекс по имени без учёта регистра
create unique index IF NOT EXISTS idx_profiles_name_unique on public.profiles using btree (lower(name));
